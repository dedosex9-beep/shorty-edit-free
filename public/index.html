const express = require('express'); const multer = require('multer'); const fs = require('fs'); const path = require('path'); const { spawn } = require('child_process'); const ffmpegPath = require('ffmpeg-static'); const app = express(); const PORT = process.env.PORT || 3000; const rootDir = __dirname; const uploadsDir = path.join(rootDir, 'uploads'); const outputsDir = path.join(rootDir, 'outputs'); const jobsDir = path.join(rootDir, 'jobs'); fs.mkdirSync(uploadsDir, { recursive: true }); fs.mkdirSync(outputsDir, { recursive: true }); fs.mkdirSync(jobsDir, { recursive: true }); const jobs = new Map(); const storage = multer.diskStorage({ destination: uploadsDir, filename: (req, file, cb) => { const cleanName = file.originalname.replace(/\s+/g, '_'); cb(null, `${Date.now()}-${cleanName}`); }, }); const upload = multer({ storage, limits: { fileSize: 500 * 1024 * 1024 }, fileFilter: (req, file, cb) => { const isImage = file.mimetype.startsWith('image/'); const isAudio = file.mimetype.startsWith('audio/'); if (isImage || isAudio) return cb(null, true); cb(new Error('Only image and audio files are allowed.')); }, }); app.use(express.static(path.join(rootDir, 'public'))); app.use('/outputs', express.static(outputsDir)); app.use('/uploads', express.static(uploadsDir)); app.get('/api/health', (req, res) => { res.json({ ok: true, app: 'Shorty Edit Free', status: 'healthy' }); }); app.post('/api/create-reel', upload.fields([ { name: 'images', maxCount: 50 }, { name: 'audio', maxCount: 1 }, ]), (req, res) => { try { const uploadedImages = req.files?.images || []; const uploadedAudio = req.files?.audio?.[0] || null; const prompt = req.body.prompt || 'Create a professional reel from my mockups.'; const style = req.body.style || 'Cinematic'; if (!uploadedImages.length) { return res.status(400).json({ ok: false, error: 'Please upload at least one image.' }); } const jobId = `job_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`; const jobDir = path.join(jobsDir, jobId); const imageDir = path.join(jobDir, 'images'); fs.mkdirSync(imageDir, { recursive: true }); uploadedImages .slice() .sort((a, b) => a.originalname.localeCompare(b.originalname)) .forEach((file, index) => { const ext = path.extname(file.originalname) || '.png'; const targetName = `frame_${String(index + 1).padStart(2, '0')}${ext}`; const targetPath = path.join(imageDir, targetName); fs.copyFileSync(file.path, targetPath); fs.unlinkSync(file.path); }); if (uploadedAudio) { const audioTarget = path.join(jobDir, 'audio' + path.extname(uploadedAudio.originalname)); fs.copyFileSync(uploadedAudio.path, audioTarget); fs.unlinkSync(uploadedAudio.path); } const outputFile = path.join(outputsDir, `${jobId}.mp4`); const job = { id: jobId, status: 'processing', progress: 5, prompt, style, outputUrl: null, createdAt: new Date().toISOString(), imageCount: uploadedImages.length, hasAudio: !!uploadedAudio, error: null, }; jobs.set(jobId, job); const ffmpegArgs = [ '-y', '-framerate', '1/3', '-pattern_type', 'glob', '-i', path.join(imageDir, '*.png').replace(/\\/g, '/'), '-vf', 'scale=1080:1920:force_original_aspect_ratio=decrease,pad=1080:1920:(ow-iw)/2:(oh-ih)/2,setsar=1,format=yuv420p,fps=30', '-c:v', 'libx264', '-pix_fmt', 'yuv420p', '-movflags', '+faststart', outputFile, ]; if (uploadedAudio) { ffmpegArgs.splice(0, 0, '-y'); const audioPath = path.join(jobDir, 'audio' + path.extname(uploadedAudio.originalname)); ffmpegArgs.splice(0, 0, '-i', audioPath); // Rebuild args with audio included after image sequence const audioArgs = [ '-y', '-framerate', '1/3', '-pattern_type', 'glob', '-i', path.join(imageDir, '*.png').replace(/\\/g, '/'), '-i', audioPath, '-filter_complex', '[0:v]scale=1080:1920:force_original_aspect_ratio=decrease,pad=1080:1920:(ow-iw)/2:(oh-ih)/2,setsar=1,format=yuv420p[f0];[f0]fps=30[v0];[1:a]aformat=sample_rates=48000:channel_layouts=stereo[a1];[v0][a1]concat=n=1:v=1:a=1[outv][outa]', '-map', '[outv]', '-map', '[outa]', '-shortest', '-c:v', 'libx264', '-pix_fmt', 'yuv420p', '-c:a', 'aac', '-movflags', '+faststart', outputFile, ]; ffmpegArgs.length = 0; audioArgs.forEach((arg) => ffmpegArgs.push(arg)); } const ffmpeg = spawn(ffmpegPath, ffmpegArgs, { stdio: ['ignore', 'pipe', 'pipe'] }); ffmpeg.stderr.on('data', (chunk) => { const text = chunk.toString(); const match = text.match(/time=(\d{2}:\d{2}:\d{2}\.\d+)/); if (match) { const timeValue = match[1]; const parts = timeValue.split(':').map(Number); const seconds = parts[0] * 3600 + parts[1] * 60 + parts[2]; const approx = Math.min(95, Math.max(10, Math.round((seconds / 30) * 100))); job.progress = approx; } }); ffmpeg.on('close', (code) => { const current = jobs.get(jobId); if (!current) return; if (code === 0 && fs.existsSync(outputFile)) { current.status = 'completed'; current.progress = 100; current.outputUrl = `/outputs/${path.basename(outputFile)}`; } else { current.status = 'failed'; current.error = 'Video rendering failed.'; } }); res.json({ ok: true, jobId, status: 'processing', progress: 5, message: 'Creating your reel...' }); } catch (error) { console.error(error); res.status(500).json({ ok: false, error: error.message || 'Unexpected server error.' }); } }); app.get('/api/jobs/:jobId', (req, res) => { const job = jobs.get(req.params.jobId); if (!job) { return res.status(404).json({ ok: false, error: 'Job not found.' }); } res.json({ ok: true, job }); }); app.get('*', (req, res) => { res.sendFile(path.join(rootDir, 'public', 'index.html')); }); app.listen(PORT, () => { console.log(`Shorty Edit Free is running on http://localhost:${PORT}`); console.log(`FFmpeg path: ${ffmpegPath}`); });
